cmake_minimum_required(VERSION 3.16)
project(paramedic_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP)
add_executable(paramedic_engine src/main.cpp)
if(OpenMP_CXX_FOUND)
  target_link_libraries(paramedic_engine PRIVATE OpenMP::OpenMP_CXX)
endif()
